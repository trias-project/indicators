This document describes the pre-processing of occurrence data of alien species in Belgium. The outputs of this pipeline will be used as input for modelling.

1 Setup

Load libraries:

library(tidyverse) # To do data science
library(here) # To find files
library(rgbif) # To get taxa from publisehd unified checklist
# pak::pak("onestop-project/elodea")
library(elodea) # To work with GBIF checklists

2 Intro

In this pipeline we create the GBIF species occurrence cube for Belgian alien species. This requires the following steps: 1. Get the alien taxa from the Global Register of Introduced and Invasive Species - Belgium. Notice that the alien taxa have different taxonomi rank. 2. Create occurrence cubes for each taxonomic rank of the alien taxa. 3. Create a cube at class level of all classes related to the alien taxa. This cube will be used to compensate the research effort bias. 3. Merge the cubes and save the cubes as interim products.

3 Get taxa

Get taxa from the Global Register of Introduced and Invasive Species - Belgium and their match to the GBIF Backbone:

taxa <- elodea::get_taxa(datasetKey = "6d9e952f-948c-4483-9807-575348147c7e")

We don’t need to retrieve distributions to select alien species as all taxa are by definition introduced in Belgium and/or one of its regions.

The columns accepted* are very important as they contain information from the GBIF Backbone:

taxa %>%
  dplyr::select(starts_with("accepted")) %>%
  head()

Taxonomic rank distribution:

taxa %>%
  select(starts_with("accepted")) %>%
  distinct() %>%
  count(acceptedTaxonRank) %>%
  arrange(desc(n))

3.1 Add class information

Add two columns, acceptedClassKey and acceptedClass to know which class the accepted taxa belong to. This information will be important later to create a cube for those classes to try to correct the research effort bias.

classes <- purrr::map(
  taxa$acceptedKey,
  function(k) {
    class_info <- rgbif::name_usage(key = k) %>%
      purrr::pluck("data") %>%
      dplyr::select(any_of(c("key", "classKey", "class")))
    if (all(c("classKey", "class") %in% names(class_info))) {
      return(class_info)
    } else {
      return(dplyr::tibble(key = class_info$key, class = NA_character_, classKey = NA_real_))
    }
  }
) %>%
  list_rbind() %>%
  distinct()
taxa <- taxa %>%
  dplyr::left_join(
    classes,
    dplyr::join_by(acceptedKey == key)
  ) %>%
  # Add prefix `accepted` as in taxa for GBIF Backbone info
  dplyr::rename(
    acceptedClass = class,
    acceptedClassKey = classKey
  ) %>%
  # Place kingdom info after class
  dplyr::relocate(
    acceptedKingdom,
    .after = last_col()
  )

4 Buid SQL query to create occurrence cubes

We need to create selveral cubes as the taxonomic rank is not uniform. The following cubes are needed: - cube at species level for species (acceptedTaxonRank = SPECIES) - cube at genus level for genera (acceptedTaxonRank = GENUS) - cube at family level for families (acceptedTaxonRank = FAMILY) - cube at taxonomic level for subtaxa, forms and varieties (acceptedTaxonRank one of (SUBSPECIES, VARIETY, FORM))

We build the list of keys needed to create the cubes:

ranks <- list(
  species = "SPECIES",
  genus = "GENUS",
  family = "FAMILY",
  taxa = list(
    subspecies = "SUBSPECIES",
    variety = "VARIETY",
    form = "FORM"
  )
)
keys <- ranks %>%
  purrr::map(
    function(x) {
      taxa %>%
        dplyr::filter(acceptedTaxonRank %in% x) %>%
        dplyr::distinct(acceptedKey) %>%
        pull()
    }
  )

We define the SQL queries needed to create an occurrence cube for the alien species in Belgium.

4.1 Occurrence cube at species level

sql_query <- "
  SELECT
    \"year\",
    GBIF_EEARGCode(
      1000,
      decimalLatitude,
      decimalLongitude,
      COALESCE(coordinateUncertaintyInMeters, 500)
    ) AS eeaCellCode,
    speciesKey,
    species,
    COUNT(*) AS occurrences,
    MIN(COALESCE(coordinateUncertaintyInMeters, 500)) AS minCoordinateUncertaintyInMeters
  FROM
    occurrence
  WHERE
    occurrenceStatus = 'PRESENT'
    AND NOT occurrence.basisofrecord IN ('FOSSIL_SPECIMEN', 'LIVING_SPECIMEN')
    AND speciesKey IN ({keys})
    AND speciesKey IS NOT NULL
    AND countryCode = 'BE'
    AND \"year\" >= 1980
    AND \"year\" IS NOT NULL
    AND hasCoordinate = TRUE
    AND NOT GBIF_STRINGARRAYCONTAINS(issue, 'ZERO_COORDINATE', TRUE)
    AND NOT GBIF_STRINGARRAYCONTAINS(issue, 'COORDINATE_OUT_OF_RANGE', TRUE)
    AND NOT GBIF_STRINGARRAYCONTAINS(issue, 'COORDINATE_INVALID', TRUE)
    AND NOT GBIF_STRINGARRAYCONTAINS(issue, 'COUNTRY_COORDINATE_MISMATCH', TRUE)
    AND NOT GBIF_STRINGARRAYCONTAINS(issue, 'TAXON_MATCH_FUZZY', TRUE)
    AND (
      LOWER(identificationVerificationStatus) NOT IN (
        'unverified',
        'unvalidated',
        'not validated',
        'under validation',
        'not able to validate',
        'control could not be conclusive due to insufficient knowledge',
        'uncertain',
        'unconfirmed',
        'unconfirmed - not reviewed',
        'validation requested'
        )
      OR identificationVerificationStatus IS NULL
    )
  GROUP BY
    \"year\",
    eeaCellCode,
    speciesKey,
    species,
    classKey
  ORDER BY
    \"year\" DESC,
    eeaCellCode ASC,
    speciesKey ASC;
"

query_filled <- glue::glue(
  sql_query,
  keys = paste(keys$species, collapse = ", ")
)

downloads <- list(

  cube_zip_files,
  rgbif::occ_download_import,

)

6 Merge the cubes

The cube containing the baseline information at class level must not be merged with the others, but set apart:

baseline_cube <- cubes$class
cubes$class <- NULL

6.1 Standardise column names

The cubes have different column names for idenitifying the taxon columns, e.g. specieskey and species, genuskey and genus. We standardise the names of the taxon columns to taxonKey and taxonName for all cubes:

# First rename the taxonomic columns of the cube at taxonomic level as it has scientificName in place of taxon
cubes$taxon <- cubes$taxon %>%
  dplyr::rename(
    taxon = scientificname
  )
# Rename all the cubes at once
cubes <- purrr::imap(
  cubes,
  function(cube, rank) {
    cube %>%
      dplyr::rename(
        taxonKey = paste0(rank, "key"),
        taxonName = rank
      )
  }
)
